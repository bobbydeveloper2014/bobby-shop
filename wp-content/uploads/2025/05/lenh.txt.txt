sudo apt update
sudo apt install -y qemu-kvm libvirt-daemon-system libvirt-clients bridge-utils virt-manager qemu-utils
# Ví dụ: Windows 10 ISO + VirtIO driver
wget -O windows.iso "https://software-download.microsoft.com/pr/Win10_22H2_English_x64.iso"
wget -O virtio-win.iso "https://fedorapeople.org/groups/virt/virtio-win/direct-downloads/archive-virtio/virtio-win-0.1.229.iso"
qemu-img create -f qcow2 windows10.qcow2 100G
sudo qemu-system-x86_64 \
  -enable-kvm \
  -m 4G \
  -cpu host \
  -smp 4 \
  -machine q35,accel=kvm \
  -drive file=windows10.qcow2,format=qcow2 \
  -cdrom Win10_22H2_English_x64.iso \
  -boot d \
  -device virtio-net-pci,netdev=net0 \
  -netdev user,id=net0 \
  -device virtio-balloon \
  -device virtio-rng-pci \
  -monitor stdio \
  -vnc :2
kết nối vm windows bằng:
localhost:5901
khi cài xong bạn chỉ cần bật lại vm bang:
sudo qemu-system-x86_64 \
  -enable-kvm \
  -m 4G \
  -cpu host \
  -smp 4 \
  -machine type=q35,accel=kvm \
  -drive file=windows10.qcow2,if=none,id=hd,format=qcow2 \
  -device virtio-blk-pci,drive=hd \
  -boot c \
  -vga std \
  -vnc :2 \
  -device virtio-net-pci,netdev=net0 \
  -netdev user,id=net0 \
  -device virtio-balloon \
  -device virtio-rng-pci \
  -monitor stdio
kết nối vm windows bằng:
localhost:5901
